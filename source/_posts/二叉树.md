### 深度优先搜索

前序，中序，后序：递归和迭代法

```Python
import sys
from pathlib import Path

# 将上一级目录（即包含 define.py 的目录）加入系统路径
sys.path.append(str(Path(__file__).resolve().parent.parent))

from define import root,TreeNode

"""
前序遍历迭代法顺序：根左右
迭代法使用栈，栈的顺序为：右左根
"""
def iterative(root: TreeNode):
    result,stack = [],[]
    curr = root
    while curr or stack:
        while curr:
            result.append(curr.val)
            stack.append(curr)
            curr = curr.left
        curr = stack.pop()
        curr = curr.right
    return result

if __name__ == '__main__':
    print(iterative(root))
```

```Python
import sys
from pathlib import Path

# 将上一级目录（即包含 define.py 的目录）加入系统路径
sys.path.append(str(Path(__file__).resolve().parent.parent))

from define import root,TreeNode

"""
后序遍历迭代法顺序：左右根

我们知道前序遍历是“根 -> 左 -> 右”。
如果我们在前序遍历时，先遍历右子树，再遍历左子树，结果就变成了“根 -> 右 -> 左”。
最后，我们把这个结果反转一下，就刚好是后序遍历的“左 -> 右 -> 根”。
"""
def iterative_postorder(root: TreeNode):
    result, stack = [], []
    curr = root
    while curr or stack:
        while curr:
            result.append(curr.val)      # 访问根节点
            stack.append(curr)
            curr = curr.right            # 先一路向右
        curr = stack.pop()
        curr = curr.left                 # 再向左
    return result[::-1]                  # 最后将结果反转
if __name__ == '__main__':
    print(iterative_postorder(root))
```

### 广度优先搜索

层序：迭代法 ：队列

```python
import sys
from pathlib import Path
from collections import deque
# 将上一级目录（即包含 define.py 的目录）加入系统路径
sys.path.append(str(Path(__file__).resolve().parent.parent))

from define import root,TreeNode

def levelOrder(root: TreeNode | None) -> list[list[int]]:
        if not root:
            return []
        result = []
        q = deque()
        q.append(root)
        while q:
            level = []  #用来保存一层的数据
            for _ in range(len(q)):  # 核心函数，分层
                node = q.popleft()
                level.append(node.val)
                if node.left:
                    q.append(node.left)
                if node.right:
                    q.append(node.right)
            result.append(level)
        return result

if __name__ == '__main__':
    print(levelOrder(root))
```

class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

# 递归

```Python
import sys
from pathlib import Path

# 将上一级目录（即包含 define.py 的目录）加入系统路径
sys.path.append(str(Path(__file__).resolve().parent.parent))

from define import root,TreeNode

"""
前序遍历递归法顺序：根左右
递归结束条件：当根节点为None时结束递归
"""
def recursive(root: TreeNode):
    result = []
    def dfs(root: TreeNode):
        if root is None:
            return
        result.append(root.val)
        dfs(root.left)
        dfs(root.right)
    dfs(root)
    return result

if __name__ == '__main__':
    print(recursive(root))
```
